# zerobus-public

DATABRICKS_CLIENT_ID=<your-service-principal-client-id>
DATABRICKS_CLIENT_SECRET=<your-service-principal-secret>

Note that you need to give the following permissions to the principal:
On catalog: USE_CATALOG.
On schema: USE_SCHEMA.
On table: MODIFY, SELECT.
Since you probably don't have permissions to grant permissions on e.g. shinkansen.default.<table>, I'd advise to create your own catalog.

Here is also my minimal Python example code for US west 2 endpoint on staging and Dogfood WS (you can also find something similar in the documentation):
import logging
import os

import record_pb2

from zerobus_sdk import ZerobusSdk, TableProperties, StreamConfigurationOptions, get_zerobus_token

sdk_logger = logging.getLogger("zerobus_sdk")
sdk_logger.setLevel(logging.INFO)
console_handler = logging.StreamHandler()

formatter = logging.Formatter('%(name)s - %(levelname)s - %(message)s')
console_handler.setFormatter(formatter)
sdk_logger.addHandler(console_handler)

# Configuration - Updated to production workspace
# Note: This reference file shows example configuration from staging environment
# Current production configuration:
# - Workspace: https://e2-demo-field-eng.cloud.databricks.com
# - Workspace ID: 1444828305810485
# - Endpoint: 1444828305810485.zerobus.eastus2.cloud.databricks.com

SERVER_ENDPOINT = "6051921418418893.zerobus.us-west-2.staging.cloud.databricks.com"  # Old staging example
DATABRICKS_WORKSPACE_URL = "https://e2-dogfood.staging.cloud.databricks.com"  # Old staging example

# Table details
TABLE_NAME = "shinkansen.default.air_quality_managed"


def create_stream():
    """
    Creates a stream using the Zerobus SDK with OAuth authentication.
    """
    sdk_handle = ZerobusSdk(SERVER_ENDPOINT)

    client_id = os.getenv("DATABRICKS_CLIENT_ID")
    client_secret = os.getenv("DATABRICKS_CLIENT_SECRET")
    if not all([client_id, client_secret]):
        raise ValueError(
            "DATABRICKS_CLIENT_ID and DATABRICKS_CLIENT_SECRET environment variables are required for UC table"
        )

    options = StreamConfigurationOptions(
        max_inflight_records=15000,
        token_factory=lambda: get_zerobus_token(
            TABLE_NAME,
            SERVER_ENDPOINT.split(".")[0],
            DATABRICKS_WORKSPACE_URL,
            client_id,
            client_secret,
        ),
    )

    table_properties = TableProperties(TABLE_NAME, record_pb2.Record.DESCRIPTOR)
    
    print(f"Creating stream for table {TABLE_NAME}")
    stream = sdk_handle.create_stream(table_properties, options)
    print(f"Stream created for table {TABLE_NAME}")

    for i in range(10000):
        ack = stream.ingest_record(record_pb2.Record(device_name=f"my_device_{i}", temp=17, humidity=42))
        if i % 1000 == 0:
            print(f"Sent {i} records to ingest")
            ack.wait_for_ack()

    print(f"Closing stream for table {TABLE_NAME}")
    stream.close()
    print(f"Stream closed for table {TABLE_NAME}")


if __name__ == "__main__":
    create_stream()

import zerobus_test_pb2
from fastapi import FastAPI
from zerobus_sdk import TableProperties
from zerobus_sdk.aio import ZerobusSdk

app = FastAPI()


ingest_endpoint = "3692013451947905.ingest.staging.cloud.databricks.com"

sdk_handle = ZerobusSdk(
    ingest_endpoint,
    "https://dbc-e2f0eb31-2b0e.staging.cloud.databricks.com",
    "<token>",
)


@app.post("/devices/")
async def create_device_reading(device_name: str, temp: int, humidity: int):
    TABLE_NAME = f"grant_doyle.zerobus.{device_name}"

    table_properties = TableProperties(TABLE_NAME, zerobus_test_pb2.testing.DESCRIPTOR)
    stream = await sdk_handle.create_stream(table_properties)

    await stream.ingest_record(
        zerobus_test_pb2.testing(device_name=device_name, temp=temp, humidity=humidity)
    )
    await stream.flush()
    await stream.close()
    return {"message": "Device reading created", "device": device_name}


if __name__ == "__main__":
    import uvicorn

    uvicorn.run(app, host="0.0.0.0", port=8000)

one thing, you'll need the create tables to have a few table properties:
CREATE OR REPLACE TABLE device_two (
    device_name STRING,
    temp INT,
    humidity INT
)
TBLPROPERTIES (
    'delta.autoOptimize.optimizeWrite' = 'true',
    'delta.autoOptimize.autoCompact' = 'true',
    'delta.enableRowTracking' = 'false'
);
and then you could route to tables like this:
import zerobus_test_pb2

    ingest_endpoint,
    "https://dbc-e2f0eb31-2b0e.staging.cloud.databricks.com",
    "<token>",
)
catalog = "grant_doyle"
schema = "zerobus"


@app.post("/devices/")
async def create_device_reading(device_name: str, temp: int, humidity: int):

        table_name = "device_one"
    elif device_name == "device_two":
        table_name = "device_two"

    )
    stream = await sdk_handle.create_stream(table_properties)

    await stream.ingest_record(
        zerobus_test_pb2.testing(device_name=device_name, temp=temp, humidity=humidity)
    )
    await stream.flush()
    await stream.close()
    return {"message": "Device reading created", "device": device_name}
